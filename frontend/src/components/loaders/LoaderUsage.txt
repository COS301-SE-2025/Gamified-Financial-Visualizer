// This is the example use of how the loader will be used in other files

import PacmanLoader from '../../components/loaders/PacmanLoader';

const Dashboard = () => {
  const [isLoading, setIsLoading] = useState(true);



// Use effect for the timing of the loader
  useEffect(() => {
    // Simulate page load (replace with real loading logic if needed)
    const timeout = setTimeout(() => {
      setIsLoading(false);
    }, 1500); // Adjust time as needed

    return () => clearTimeout(timeout);
  }, []);


// at the top of every main conatiner 
  return (
    <div className="relative h-screen bg-gray-50">
      {/* Loader overlay */}
      {isLoading && (
        <div className="absolute inset-0 z-50 flex items-center justify-center bg-white">
          <PacmanLoader />
        </div>
      )}
    </div>
  );
}
